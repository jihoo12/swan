cmake_minimum_required(VERSION 3.20)
project(swan_headless_example LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(swan 0.18 CONFIG REQUIRED)
add_executable(garden-stats main.cpp)
target_link_libraries(garden-stats PRIVATE swan::headless)
target_compile_definitions(garden-stats PRIVATE SWAN_SCENES="${SWAN_DATA_DIR}/assets/scenes")
install(TARGETS garden-stats)
