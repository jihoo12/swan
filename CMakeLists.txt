cmake_minimum_required(VERSION 3.20)
project(swan VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Vulkan REQUIRED)
find_package(glfw3 3.4 REQUIRED)
find_package(glm CONFIG REQUIRED)
find_program(GLSLANG_VALIDATOR glslangValidator REQUIRED)
include(GNUInstallDirs)
foreach(stage vert frag)
  set(output "${CMAKE_CURRENT_BINARY_DIR}/shaders/scene.${stage}.spv")
  add_custom_command(OUTPUT "${output}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
    COMMAND ${GLSLANG_VALIDATOR} -V --target-env vulkan1.3 "${CMAKE_CURRENT_SOURCE_DIR}/shaders/scene.${stage}" -o "${output}"
    DEPENDS "shaders/scene.${stage}" VERBATIM)
  list(APPEND SHADERS "${output}")
endforeach()
add_custom_target(swan_shaders ALL DEPENDS ${SHADERS})
add_executable(swan src/main.cpp src/engine.cpp)
add_dependencies(swan swan_shaders)
target_link_libraries(swan PRIVATE Vulkan::Vulkan glfw glm::glm)
target_compile_definitions(swan PRIVATE GLM_FORCE_DEPTH_ZERO_TO_ONE GLM_FORCE_RADIANS SWAN_SHADER_DIR="${CMAKE_CURRENT_BINARY_DIR}/shaders")
target_compile_options(swan PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
install(TARGETS swan RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES ${SHADERS} DESTINATION ${CMAKE_INSTALL_DATADIR}/swan/shaders)
enable_testing()
add_executable(scene_test tests/scene_test.cpp)
target_link_libraries(scene_test PRIVATE glm::glm)
target_include_directories(scene_test PRIVATE src)
add_test(NAME scene COMMAND scene_test)
