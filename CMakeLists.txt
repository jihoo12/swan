cmake_minimum_required(VERSION 3.20)
project(swan VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Vulkan REQUIRED)
find_package(glfw3 3.4 REQUIRED)
find_package(glm CONFIG REQUIRED)
find_program(GLSLANG_VALIDATOR glslangValidator REQUIRED)
include(GNUInstallDirs)
foreach(stage vert frag)
  set(output "${CMAKE_CURRENT_BINARY_DIR}/shaders/scene.${stage}.spv")
  add_custom_command(OUTPUT "${output}"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/shaders"
    COMMAND ${GLSLANG_VALIDATOR} -V --target-env vulkan1.3 "${CMAKE_CURRENT_SOURCE_DIR}/shaders/scene.${stage}" -o "${output}"
    DEPENDS "shaders/scene.${stage}" VERBATIM)
  list(APPEND SHADERS "${output}")
endforeach()
add_custom_target(swan_shaders ALL DEPENDS ${SHADERS})
add_library(swan_core src/scene.cpp src/physics.cpp)
target_include_directories(swan_core PUBLIC src)
target_link_libraries(swan_core PUBLIC glm::glm)
target_compile_definitions(swan_core PUBLIC GLM_FORCE_DEPTH_ZERO_TO_ONE GLM_FORCE_RADIANS)
target_compile_options(swan_core PRIVATE -Wall -Wextra -Wpedantic)
add_library(swan_demo src/garden.cpp src/game.cpp)
target_link_libraries(swan_demo PUBLIC swan_core)
target_compile_options(swan_demo PRIVATE -Wall -Wextra -Wpedantic)
add_library(swan_renderer src/renderer.cpp)
add_dependencies(swan_renderer swan_shaders)
target_link_libraries(swan_renderer PUBLIC swan_core Vulkan::Vulkan glfw)
target_compile_definitions(swan_renderer PRIVATE SWAN_SHADER_DIR="${CMAKE_CURRENT_BINARY_DIR}/shaders")
target_compile_options(swan_renderer PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
add_library(swan_runtime src/engine.cpp)
target_link_libraries(swan_runtime PUBLIC swan_renderer)
target_compile_options(swan_runtime PRIVATE -Wall -Wextra -Wpedantic)
add_executable(swan src/main.cpp)
target_link_libraries(swan PRIVATE swan_runtime swan_demo)
target_compile_options(swan PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS swan RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES ${SHADERS} DESTINATION ${CMAKE_INSTALL_DATADIR}/swan/shaders)
enable_testing()
add_executable(scene_test tests/scene_test.cpp)
target_link_libraries(scene_test PRIVATE swan_demo)
target_include_directories(scene_test PRIVATE src)
add_test(NAME scene COMMAND scene_test)
add_executable(engine_test tests/engine_test.cpp)
target_link_libraries(engine_test PRIVATE swan_demo)
add_test(NAME engine COMMAND engine_test)
